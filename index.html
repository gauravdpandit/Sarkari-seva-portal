from bs4 import BeautifulSoup
import google.generativeai as genai

GEMINI_API_KEY = os.environ.get("GEMINI_API_KEY", "")
TELEGRAM_BOT_TOKEN = os.environ.get("TELEGRAM_BOT_TOKEN", "")
TELEGRAM_CHAT_ID = os.environ.get("TELEGRAM_CHAT_ID", "")
FORM_BOOKING_URL = os.environ.get("FORM_BOOKING_URL", "https://t.me/YourBot")

RSS_FEEDS = [
    "https://pib.gov.in/RssMain.aspx?ModId=6&Lang=2&Regid=3",
    "https://news.google.com/rss/search?q=Bihar+Sarkari+Yojana+Scholarship+Form&hl=hi&gl=IN&ceid=IN:hi"
]

def summarize_with_gemini(title, summary, link):
    genai.configure(api_key=GEMINI_API_KEY)
    model = genai.GenerativeModel("gemini-1.5-flash")
    prompt = f"सरल हिंदी में सरकारी योजना/फॉर्म अलर्ट पोस्ट बनाएं:\nशीर्षक: {title}\nविवरण: {summary}"
    return model.generate_content(prompt).text.strip()

def send_to_telegram(text, source_link):
    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
    buttons = {"inline_keyboard": [
        [{"text": "📝 घर बैठे फॉर्म भरवाएं (₹50)", "url": FORM_BOOKING_URL}],
        [{"text": "🔗 आधिकारिक लिंक", "url": source_link}]
    ]}
    requests.post(url, data={
        "chat_id": TELEGRAM_CHAT_ID,
        "text": text,
        "reply_markup": json.dumps(buttons)
    }, timeout=20)
